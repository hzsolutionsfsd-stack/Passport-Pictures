<script>
  (function() {
    // ---------- DOM elements ----------
    const video = document.getElementById('video');
    const canvas = document.getElementById('previewCanvas');
    const ctx = canvas.getContext('2d');
    const admissionInput = document.getElementById('admissionNo');
    const captureBtn = document.getElementById('captureBtn');
    const saveBtn = document.getElementById('saveBtn');
    const switchBtn = document.getElementById('switchCameraBtn');
    const statusMsg = document.getElementById('statusMsg');
    const permissionOverlay = document.getElementById('permissionOverlay');
    const allowCameraBtn = document.getElementById('allowCameraBtn');

    // ---------- State ----------
    let stream = null;
    let currentFacingMode = 'user';
    let capturedImageData = null;
    let permissionGranted = false;

    const OUTPUT_SIZE = 300;

    // ---------- Helper: show status ----------
    function setStatus(text, isError = false) {
      statusMsg.textContent = text;
      statusMsg.style.color = isError ? '#b91c1c' : '#666';
    }

    // ---------- Enable/disable UI ----------
    function setUIEnabled(enabled) {
      captureBtn.disabled = !enabled;
      switchBtn.disabled = !enabled;
    }

    // ---------- Request camera permission with explicit user gesture ----------
    async function requestCameraPermission() {
      allowCameraBtn.disabled = true;
      allowCameraBtn.textContent = 'Requesting...';

      try {
        // This call triggers the browser's native permission prompt
        const tempStream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: currentFacingMode },
          audio: false
        });

        // Permission granted - stop temp stream
        tempStream.getTracks().forEach(track => track.stop());
        permissionGranted = true;
        permissionOverlay.classList.add('hidden');
        setStatus('✅ Camera permission granted');

        // Now start the actual camera
        await startCamera(currentFacingMode);
        setUIEnabled(true);

      } catch (err) {
        console.error('Permission error:', err);
        allowCameraBtn.disabled = false;
        allowCameraBtn.textContent = 'Allow Camera';

        if (err.name === 'NotAllowedError' || err.name === 'PermissionDeniedError') {
          setStatus('❌ Camera permission denied. Please allow camera in browser settings and reload.', true);
          permissionOverlay.querySelector('p').innerHTML =
            'Camera access was denied.<br>Please allow camera access in your browser settings, then reload the page.';
        } else if (err.name === 'NotFoundError' || err.name === 'DevicesNotFoundError') {
          setStatus('❌ No camera device found on this device.', true);
          permissionOverlay.querySelector('p').innerHTML =
            'No camera was found on this device.<br>Please connect a camera and try again.';
        } else if (err.name === 'NotReadableError' || err.name === 'TrackStartError') {
          setStatus('❌ Camera is in use by another app. Close it and try again.', true);
          permissionOverlay.querySelector('p').innerHTML =
            'The camera is being used by another application.<br>Please close it and try again.';
        } else {
          setStatus('❌ Camera error: ' + err.message, true);
        }
      }
    }

    // ---------- Start camera stream ----------
    async function startCamera(facingMode = 'user') {
      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        setStatus('❌ Camera API not supported in this browser', true);
        return false;
      }

      if (stream) {
        stream.getTracks().forEach(track => track.stop());
        stream = null;
      }

      const constraints = {
        video: {
          facingMode: facingMode,
          width: { ideal: 1280 },
          height: { ideal: 960 }
        },
        audio: false
      };

      try {
        const newStream = await navigator.mediaDevices.getUserMedia(constraints);
        stream = newStream;
        video.srcObject = newStream;
        await video.play();
        video.style.display = 'block';
        canvas.style.display = 'none';
        setStatus('Camera ready');
        return true;
      } catch (err) {
        console.error('Error starting camera:', err);
        setStatus('❌ Could not start camera: ' + err.message, true);
        return false;
      }
    }

    // ---------- Switch camera ----------
    async function switchCamera() {
      currentFacingMode = currentFacingMode === 'user' ? 'environment' : 'user';
      setStatus(`Switching to ${currentFacingMode} camera...`);
      const success = await startCamera(currentFacingMode);
      if (!success) {
        currentFacingMode = currentFacingMode === 'user' ? 'environment' : 'user';
      }
    }

    // ---------- Validate admission number ----------
    function getAdmissionNo() {
      const val = admissionInput.value.trim();
      if (/^\d{4}$/.test(val)) return val;
      return null;
    }

    // ---------- Background color ----------
    function getSelectedBgColor() {
      const selected = document.querySelector('input[name="bg"]:checked');
      if (!selected) return '#ffffff';
      switch (selected.value) {
        case 'green': return '#2e7d32';
        case 'blue': return '#b3e5fc';
        case 'white':
        default: return '#ffffff';
      }
    }

    // ---------- Hex to RGB ----------
    function hexToRgb(hex) {
      const bigint = parseInt(hex.replace('#', ''), 16);
      return {
        r: (bigint >> 16) & 255,
        g: (bigint >> 8) & 255,
        b: bigint & 255
      };
    }

    // ---------- Capture photo ----------
    function capturePhoto() {
      const admissionNo = getAdmissionNo();
      if (!admissionNo) {
        setStatus('⚠️ Enter a valid 4-digit admission number', true);
        admissionInput.focus();
        return;
      }

      if (!video.videoWidth || !video.videoHeight) {
        setStatus('⚠️ Camera not ready, please wait', true);
        return;
      }

      const rawCanvas = document.createElement('canvas');
      rawCanvas.width = video.videoWidth;
      rawCanvas.height = video.videoHeight;
      const rawCtx = rawCanvas.getContext('2d');
      rawCtx.drawImage(video, 0, 0, rawCanvas.width, rawCanvas.height);

      const targetSize = OUTPUT_SIZE;
      const size = Math.min(rawCanvas.width, rawCanvas.height);
      const offsetX = (rawCanvas.width - size) / 2;
      const offsetY = (rawCanvas.height - size) / 2;

      const cropCanvas = document.createElement('canvas');
      cropCanvas.width = size;
      cropCanvas.height = size;
      const cropCtx = cropCanvas.getContext('2d');
      cropCtx.drawImage(rawCanvas, offsetX, offsetY, size, size, 0, 0, size, size);

      const finalCanvas = document.createElement('canvas');
      finalCanvas.width = targetSize;
      finalCanvas.height = targetSize;
      const finalCtx = finalCanvas.getContext('2d');

      finalCtx.drawImage(cropCanvas, 0, 0, size, size, 0, 0, targetSize, targetSize);

      // Background replacement
      const imageData = finalCtx.getImageData(0, 0, targetSize, targetSize);
      const data = imageData.data;
      const bgColor = getSelectedBgColor();
      const bgRgb = hexToRgb(bgColor);

      for (let i = 0; i < data.length; i += 4) {
        const r = data[i];
        const g = data[i + 1];
        const b = data[i + 2];

        const lum = 0.299 * r + 0.587 * g + 0.114 * b;
        const max = Math.max(r, g, b);
        const min = Math.min(r, g, b);
        const sat = max === 0 ? 0 : (max - min) / max;

        let isBackground = false;

        if (lum > 190 && sat < 0.2) isBackground = true;
        else if (lum > 160 && sat < 0.15) isBackground = true;

        const distToWhite = Math.sqrt((r-255)**2 + (g-255)**2 + (b-255)**2);
        if (distToWhite < 60) isBackground = true;

        if (bgColor !== '#ffffff') {
          const dist = Math.sqrt((r-bgRgb.r)**2 + (g-bgRgb.g)**2 + (b-bgRgb.b)**2);
          if (dist < 80) isBackground = true;
        }

        if (lum > 230) isBackground = true;

        if (isBackground) {
          data[i] = bgRgb.r;
          data[i + 1] = bgRgb.g;
          data[i + 2] = bgRgb.b;
        }
      }

      finalCtx.putImageData(imageData, 0, 0);

      // Draw admission number at bottom
      const text = admissionNo;
      finalCtx.font = 'bold 16px "Segoe UI", system-ui, sans-serif';
      finalCtx.textAlign = 'center';
      finalCtx.textBaseline = 'bottom';

      const textWidth = finalCtx.measureText(text).width;
      const padding = 4;
      const x = targetSize / 2;
      const y = targetSize - 10;

      finalCtx.fillStyle = 'rgba(255,255,255,0.7)';
      finalCtx.fillRect(x - textWidth/2 - padding, y - 16 - padding, textWidth + padding*2, 20 + padding*2);

      finalCtx.fillStyle = '#000000';
      finalCtx.font = 'bold 16px "Segoe UI", system-ui, sans-serif';
      finalCtx.fillText(text, x, y);

      const dataUrl = finalCanvas.toDataURL('image/png');
      capturedImageData = { dataUrl, width: targetSize, height: targetSize };

      canvas.width = targetSize;
      canvas.height = targetSize;
      const displayCtx = canvas.getContext('2d');
      displayCtx.drawImage(finalCanvas, 0, 0);

      video.style.display = 'none';
      canvas.style.display = 'block';

      saveBtn.disabled = false;
      setStatus('✅ Photo captured. Ready to save.');
    }

    // ---------- Save photo ----------
    function savePhoto() {
      if (!capturedImageData) {
        setStatus('⚠️ No photo captured yet', true);
        return;
      }

      const link = document.createElement('a');
      link.href = capturedImageData.dataUrl;
      link.download = `passport_${Date.now()}.png`;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);

      setStatus('📁 Photo saved to downloads / photos folder');
    }

    // ---------- Event listeners ----------
    allowCameraBtn.addEventListener('click', requestCameraPermission);

    captureBtn.addEventListener('click', () => {
      if (canvas.style.display === 'block') {
        video.style.display = 'block';
        canvas.style.display = 'none';
        setStatus('Live camera ready. Press capture again.');
        return;
      }
      capturePhoto();
    });

    saveBtn.addEventListener('click', savePhoto);

    switchBtn.addEventListener('click', async () => {
      if (canvas.style.display === 'block') {
        video.style.display = 'block';
        canvas.style.display = 'none';
      }
      await switchCamera();
    });

    admissionInput.addEventListener('input', (e) => {
      e.target.value = e.target.value.replace(/\D/g, '').slice(0, 4);
    });

    // Cleanup
    window.addEventListener('beforeunload', () => {
      if (stream) {
        stream.getTracks().forEach(track => track.stop());
      }
    });

    video.addEventListener('error', (e) => {
      console.error('Video error:', e);
      setStatus('⚠️ Camera error', true);
    });

  })();
</script>
